# P0 — TestRail parity: execution + detail + suites + milestones + RBAC

## What this bundle fixes (from the audit)
1. Runs were dead: no way to mark Passed/Failed/Blocked/Retest. → NEW /test-runs/[id] + PATCH /api/test-executions/[id]
2. No detail/edit/delete for cases, plans, defects, requirements, projects. → NEW [id] APIs + detail pages for cases & plans
3. Suites + milestones existed in schema, never used. → NEW /api/test-suites + /api/releases (milestones)
4. RBAC defined but never enforced (first-org bug). → NEW require-org-permission.ts, enforced in all new/updated routes
5. Manual defect keys + no link from failure to defect. → defects POST auto-generates key + accepts testCaseId/testExecutionId
6. Steps were free-text. → test-cases PATCH now replaces test_case_steps; detail page has structured steps editor
7. Runs always included ALL cases. → test-runs POST now accepts testPlanId / suiteIds / caseIds / assignedTo / description / configuration

## Install
1. Run supabase_p0_migration.sql in Supabase SQL Editor.
   Run the ALTER TYPE line alone first if you get "cannot run inside transaction block":
     ALTER TYPE execution_status ADD VALUE IF NOT EXISTS 'retest';
2. Copy all files in this bundle over your repo, preserving paths.
   Overwrites: src/app/api/test-runs/route.ts, src/app/api/defects/route.ts, src/app/api/projects/[id]/route.ts, src/app/test-runs/new/page.tsx
   New: everything else.
3. Apply PATCH_LIST_PAGES.md (wrap list rows in Links).
4. Apply src/types/supabase-p0-patch.md.
5. `npm run dev` — test the checklist below.

## Test checklist (5 min)
- Create a project → create 2 test cases with steps → create a suite → add cases to suite via POST /api/test-suites/[id]/cases
- Create a milestone via POST /api/releases { projectId, versionName: "v1.0" } → create a plan linked to it
- Create a run from that plan + suite → open /test-runs/[id] → expand a case → mark Failed + comment → Save → Create Defect from this Failure → verify defect appears in /defects with test_case_id + test_execution_id
- Bulk: select 2 Not Run → Mark Passed → progress updates
- Complete Run → try to edit an execution → expect 409 → Reopen → edit works again
- As a viewer (second account): try PATCH /api/test-executions/[id] → expect 403. Try GET → expect 200.

## What P0 does NOT yet do (intentionally left for P1)
- Defect / Requirement detail UI pages (APIs exist)
- Suites UI page (API exists, used via run creation + plan detail)
- Milestones UI page (API exists, used via plan detail)
- Per-step execution results UI (API accepts stepResults)
- Notifications, audit log, custom fields, integrations

import { NextRequest, NextResponse } from 'next/server';
import { createRouteSupabaseClient } from '@/lib/supabase/route-client';
import { requireOrgPermission, permissionError, UUID_RE } from '@/lib/auth/require-org-permission';
const SEVERITIES = ['trivial','minor','major','critical','blocker'];
const PRIORITIES = ['low','medium','high','critical'];
const STATUSES = ['new','open','in_progress','fixed','ready_for_retest','reopened','closed','rejected'];
export async function GET(_req: NextRequest, { params }: { params: { id: string } }) {
  const supabase = createRouteSupabaseClient();
  const { data: { user }, error: authError } = await supabase.auth.getUser();
  if (authError || !user) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  if (!UUID_RE.test(params.id)) return NextResponse.json({ error: 'Defect not found.' }, { status: 404 });
  const { data: d, error } = await supabase.from('defects').select('*, test_cases(case_key,title), test_executions(id,status)').eq('id', params.id).maybeSingle();
  if (error || !d) return NextResponse.json({ error: 'Defect not found.' }, { status: 404 });
  try { await requireOrgPermission(supabase, user.id, (d as any).organization_id, 'canViewReports'); } catch (e) { const pe = permissionError(e); return NextResponse.json({ error: pe.message }, { status: pe.status }); }
  return NextResponse.json({ defect: d });
}
export async function PATCH(request: NextRequest, { params }: { params: { id: string } }) {
  const supabase = createRouteSupabaseClient();
  const { data: { user }, error: authError } = await supabase.auth.getUser();
  if (authError || !user) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  let body: Record<string, unknown>;
  try { body = await request.json(); } catch { return NextResponse.json({ error: 'Request body must be valid JSON.' }, { status: 400 }); }
  const { data: d } = await supabase.from('defects').select('id, organization_id, project_id').eq('id', params.id).maybeSingle();
  if (!d) return NextResponse.json({ error: 'Defect not found.' }, { status: 404 });
  try { await requireOrgPermission(supabase, user.id, (d as any).organization_id, 'canEditDefect'); } catch (e) { const pe = permissionError(e); return NextResponse.json({ error: pe.message }, { status: pe.status }); }
  const patch: Record<string, unknown> = {};
  if (typeof body.summary === 'string' && body.summary.trim()) patch.summary = body.summary.trim();
  if (typeof body.description === 'string') patch.description = body.description.trim() || null;
  if (typeof body.severity === 'string' && SEVERITIES.includes(body.severity)) patch.severity = body.severity;
  if (typeof body.priority === 'string' && PRIORITIES.includes(body.priority)) patch.priority = body.priority;
  if (typeof body.status === 'string' && STATUSES.includes(body.status)) patch.status = body.status;
  if (typeof body.environment === 'string') patch.environment = body.environment.trim() || null;
  if (typeof body.stepsToReproduce === 'string') patch.steps_to_reproduce = body.stepsToReproduce.trim() || null;
  if (typeof body.expectedResult === 'string') patch.expected_result = body.expectedResult.trim() || null;
  if (typeof body.actualResult === 'string') patch.actual_result = body.actualResult.trim() || null;
  if (typeof body.assignedTo === 'string') patch.assigned_to = body.assignedTo || null;
  if (typeof body.testCaseId === 'string') patch.test_case_id = body.testCaseId || null;
  if (typeof body.testExecutionId === 'string') patch.test_execution_id = body.testExecutionId || null;
  if (!Object.keys(patch).length) return NextResponse.json({ error: 'Nothing to update.' }, { status: 400 });
  // assigned user must be org member
  if (patch.assigned_to) {
    const { data: m } = await supabase.from('organization_members').select('user_id').eq('organization_id', (d as any).organization_id).eq('user_id', patch.assigned_to as string).maybeSingle();
    if (!m) return NextResponse.json({ error: 'Assignee must be a member of this organization.' }, { status: 400 });
  }
  const { data: updated, error } = await supabase.from('defects').update(patch).eq('id', params.id).select().single();
  if (error) return NextResponse.json({ error: error.message }, { status: 400 });
  return NextResponse.json({ defect: updated });
}
export async function DELETE(_req: NextRequest, { params }: { params: { id: string } }) {
  const supabase = createRouteSupabaseClient();
  const { data: { user }, error: authError } = await supabase.auth.getUser();
  if (authError || !user) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  const { data: d } = await supabase.from('defects').select('id, organization_id').eq('id', params.id).maybeSingle();
  if (!d) return NextResponse.json({ error: 'Defect not found.' }, { status: 404 });
  try { await requireOrgPermission(supabase, user.id, (d as any).organization_id, 'canDeleteDefect'); } catch (e) { const pe = permissionError(e); return NextResponse.json({ error: pe.message }, { status: pe.status }); }
  const { error } = await supabase.from('defects').delete().eq('id', params.id);
  if (error) return NextResponse.json({ error: error.message }, { status: 400 });
  return NextResponse.json({ deleted: true });
}

import { NextRequest, NextResponse } from 'next/server';
import { createRouteSupabaseClient } from '@/lib/supabase/route-client';
import { requireOrgPermission, permissionError } from '@/lib/auth/require-org-permission';
const severities = ['trivial','minor','major','critical','blocker'] as const;
const priorities = ['low','medium','high','critical'] as const;
export async function GET(request: NextRequest) {
  const supabase = createRouteSupabaseClient();
  const { data: { user }, error: authError } = await supabase.auth.getUser();
  if (authError || !user) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  const projectId = request.nextUrl.searchParams.get('projectId');
  let query = supabase.from('defects').select('*').order('created_at', { ascending: false });
  if (projectId) query = query.eq('project_id', projectId);
  const { data, error } = await query;
  if (error) return NextResponse.json({ error: 'Could not load defects.' }, { status: 500 });
  return NextResponse.json({ defects: data ?? [] });
}
export async function POST(request: NextRequest) {
  const supabase = createRouteSupabaseClient();
  const { data: { user }, error: authError } = await supabase.auth.getUser();
  if (authError || !user) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  let body: Record<string, unknown>;
  try { body = await request.json(); } catch { return NextResponse.json({ error: 'Request body must be valid JSON.' }, { status: 400 }); }
  const projectId = typeof body.projectId === 'string' ? body.projectId : '';
  const summary = typeof body.summary === 'string' ? body.summary.trim() : '';
  if (!projectId || !summary) return NextResponse.json({ error: 'Project and summary are required.' }, { status: 400 });
  const { data: project, error: projectError } = await supabase.from('projects').select('id, organization_id, key').eq('id', projectId).maybeSingle();
  if (projectError || !project) return NextResponse.json({ error: 'Select a project you can access.' }, { status: 400 });
  try { await requireOrgPermission(supabase, user.id, (project as any).organization_id, 'canCreateDefect'); } catch (e) { const pe = permissionError(e); return NextResponse.json({ error: pe.message }, { status: pe.status }); }
  // Auto-generate key if caller did not supply one (execution page + new defect page can omit it)
  let defectKey = typeof body.defectKey === 'string' ? body.defectKey.trim().toUpperCase() : '';
  if (!defectKey) {
    const prefix = `${(project as any).key ?? 'BUG'}-`;
    const { count } = await supabase.from('defects').select('id', { count: 'exact', head: true }).eq('project_id', projectId);
    let n = (count ?? 0) + 1;
    for (let attempt = 0; attempt < 20; attempt++) {
      const candidate = `${prefix}${n}`;
      const { data: existing } = await supabase.from('defects').select('id').eq('project_id', projectId).eq('defect_key', candidate).maybeSingle();
      if (!existing) { defectKey = candidate; break; }
      n++;
    }
    if (!defectKey) defectKey = `${prefix}${Date.now().toString().slice(-6)}`;
  }
  const severity = typeof body.severity === 'string' && severities.includes(body.severity as any) ? body.severity : 'major';
  const priority = typeof body.priority === 'string' && priorities.includes(body.priority as any) ? body.priority : 'medium';
  // Validate linked execution / case belong to this project
  let testCaseId: string | null = null, testExecutionId: string | null = null;
  if (typeof body.testCaseId === 'string' && body.testCaseId) {
    const { data: tc } = await supabase.from('test_cases').select('id').eq('id', body.testCaseId).eq('project_id', projectId).maybeSingle();
    if (!tc) return NextResponse.json({ error: 'Linked test case must belong to this project.' }, { status: 400 });
    testCaseId = tc.id;
  }
  if (typeof body.testExecutionId === 'string' && body.testExecutionId) {
    const { data: ex } = await supabase.from('test_executions').select('id, test_case_id, test_runs!inner(project_id)').eq('id', body.testExecutionId).maybeSingle();
    if (!ex) return NextResponse.json({ error: 'Linked execution not found.' }, { status: 400 });
    testExecutionId = ex.id;
    if (!testCaseId) testCaseId = (ex as any).test_case_id;
  }
  let assignedTo: string | null = null;
  if (typeof body.assignedTo === 'string' && body.assignedTo) {
    const { data: m } = await supabase.from('organization_members').select('user_id').eq('organization_id', (project as any).organization_id).eq('user_id', body.assignedTo).maybeSingle();
    if (!m) return NextResponse.json({ error: 'Assignee must be a member of this organization.' }, { status: 400 });
    assignedTo = body.assignedTo;
  }
  const { data: defect, error } = await supabase.from('defects').insert({
    project_id: project.id, organization_id: (project as any).organization_id, defect_key: defectKey, summary,
    description: typeof body.description === 'string' ? body.description.trim() || null : null,
    severity, priority, status: 'new',
    environment: typeof body.environment === 'string' ? body.environment.trim() || null : null,
    steps_to_reproduce: typeof body.stepsToReproduce === 'string' ? body.stepsToReproduce.trim() || null : null,
    expected_result: typeof body.expectedResult === 'string' ? body.expectedResult.trim() || null : null,
    actual_result: typeof body.actualResult === 'string' ? body.actualResult.trim() || null : null,
    reporter_id: user.id, assigned_to: assignedTo, test_case_id: testCaseId, test_execution_id: testExecutionId,
  }).select().single();
  if (error) return NextResponse.json({ error: error.code === '23505' ? 'That defect key is already used in this project.' : error.message }, { status: error.code === '23505' ? 409 : 400 });
  return NextResponse.json({ defect }, { status: 201 });
}

// PATCH /api/test-executions/[id] — the core TestRail action: mark result
// Body: { status, actualResult?, comments?, stepResults?: [{ stepNumber, status, actualResult }] }
export async function PATCH(request: NextRequest, { params }: { params: { id: string } }) {
  const supabase = createRouteSupabaseClient();
  const { data: { user }, error: authError } = await supabase.auth.getUser();
  if (authError || !user) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  if (!UUID_RE.test(params.id)) return NextResponse.json({ error: 'Execution not found.' }, { status: 404 });
  let body: Record<string, unknown>;
  try { body = await request.json(); } catch { return NextResponse.json({ error: 'Request body must be valid JSON.' }, { status: 400 }); }
  const status = typeof body.status === 'string' ? body.status : '';
  if (!STATUSES.includes(status as any)) return NextResponse.json({ error: 'Status must be one of: ' + STATUSES.join(', ') }, { status: 400 });

  const { data: ex, error } = await supabase.from('test_executions').select('id, organization_id, test_run_id, test_case_id, status').eq('id', params.id).maybeSingle();
  if (error || !ex) return NextResponse.json({ error: 'Execution not found.' }, { status: 404 });
  try { await requireOrgPermission(supabase, user.id, (ex as any).organization_id, 'canExecuteTests'); } catch (e) { const pe = permissionError(e); return NextResponse.json({ error: pe.message }, { status: pe.status }); }

  // Do not allow editing executions of a completed run — reopen the run first.
  const { data: run } = await supabase.from('test_runs').select('status').eq('id', (ex as any).test_run_id).maybeSingle();
  if ((run as any)?.status === 'completed' && status !== (ex as any).status) {
    return NextResponse.json({ error: 'This run is completed. Reopen the run to change results.' }, { status: 409 });
  }

  const patch: Record<string, unknown> = {
    status,
    tester_id: user.id,
    actual_result: typeof body.actualResult === 'string' ? body.actualResult.trim() || null : null,
    comments: typeof body.comments === 'string' ? body.comments.trim() || null : null,
    executed_at: status === 'not_run' ? null : new Date().toISOString(),
  };
  const { data: updated, error: updErr } = await supabase.from('test_executions').update(patch).eq('id', params.id).select('*, test_cases(case_key,title)').single();
  if (updErr) return NextResponse.json({ error: updErr.message }, { status: 400 });

  // Optional per-step results
  if (Array.isArray(body.stepResults)) {
    for (const s of body.stepResults as any[]) {
      if (!s || typeof s.stepNumber !== 'number') continue;
      if (!STATUSES.includes(s.status)) continue;
      await supabase.from('execution_steps').upsert(
        { test_execution_id: params.id, step_number: s.stepNumber, status: s.status, actual_result: typeof s.actualResult === 'string' ? s.actualResult : null },
        { onConflict: 'test_execution_id,step_number' } as any
      );
    }
  }
  return NextResponse.json({ execution: updated });
}

export async function POST(request: NextRequest) {
  const supabase = createRouteSupabaseClient();
  const { data: { user }, error: authError } = await supabase.auth.getUser();
  if (authError || !user) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  let body: Record<string, unknown>;
  try { body = await request.json(); } catch { return NextResponse.json({ error: 'Request body must be valid JSON.' }, { status: 400 }); }
  const projectId = typeof body.projectId === 'string' ? body.projectId : '';
  const name = typeof body.name === 'string' ? body.name.trim() : '';
  if (!projectId || !name) return NextResponse.json({ error: 'Project and test run name are required.' }, { status: 400 });
  const { data: project, error: projectError } = await supabase.from('projects').select('id, organization_id').eq('id', projectId).maybeSingle();
  if (projectError || !project) return NextResponse.json({ error: 'Select a project you can access.' }, { status: 400 });
  try { await requireOrgPermission(supabase, user.id, (project as any).organization_id, 'canExecuteTests'); } catch (e) { const pe = permissionError(e); return NextResponse.json({ error: pe.message }, { status: pe.status }); }

  let testPlanId: string | null = null;
  if (typeof body.testPlanId === 'string' && body.testPlanId) {
    const { data: plan } = await supabase.from('test_plans').select('id').eq('id', body.testPlanId).eq('project_id', project.id).maybeSingle();
    if (!plan) return NextResponse.json({ error: 'Select a test plan from this project.' }, { status: 400 });
    testPlanId = plan.id;
  }
  // Assignee must be a member of the org
  let assignedTo = user.id;
  if (typeof body.assignedTo === 'string' && body.assignedTo && UUID_RE.test(body.assignedTo)) {
    const { data: m } = await supabase.from('organization_members').select('user_id').eq('organization_id', (project as any).organization_id).eq('user_id', body.assignedTo).maybeSingle();
    if (m) assignedTo = body.assignedTo;
  }
  const { data: run, error: runError } = await supabase.from('test_runs').insert({
    project_id: project.id, organization_id: (project as any).organization_id, name,
    test_plan_id: testPlanId, assigned_to: assignedTo, status: 'in_progress',
    description: typeof body.description === 'string' ? body.description.trim() || null : null,
    configuration: typeof body.configuration === 'string' ? body.configuration.trim() || null : null,
  }).select().single();
  if (runError || !run) return NextResponse.json({ error: runError?.message || 'Could not create the test run.' }, { status: 400 });

  // Resolve cases: explicit caseIds > suiteIds > all cases in project
  let caseIds: string[] = [];
  const rawCaseIds = Array.isArray(body.caseIds) ? body.caseIds.filter((x): x is string => typeof x === 'string') : [];
  const rawSuiteIds = Array.isArray(body.suiteIds) ? body.suiteIds.filter((x): x is string => typeof x === 'string') : [];
  if (rawCaseIds.length) {
    const { data: cases } = await supabase.from('test_cases').select('id').eq('project_id', project.id).in('id', rawCaseIds);
    caseIds = (cases ?? []).map((c: any) => c.id);
  } else if (rawSuiteIds.length) {
    const { data: links } = await supabase.from('test_suite_cases').select('test_case_id').in('test_suite_id', rawSuiteIds);
    caseIds = [...new Set((links ?? []).map((l: any) => l.test_case_id))];
    if (caseIds.length) {
      const { data: cases } = await supabase.from('test_cases').select('id').eq('project_id', project.id).in('id', caseIds);
      caseIds = (cases ?? []).map((c: any) => c.id);
    }
  } else {
    const { data: cases } = await supabase.from('test_cases').select('id').eq('project_id', project.id);
    caseIds = (cases ?? []).map((c: any) => c.id);
  }
  if (!caseIds.length) {
    // Keep the empty run — UI shows "No cases" and user can add cases later via re-create. Do not fail.
    return NextResponse.json({ testRun: run, testCaseCount: 0 }, { status: 201 });
  }
  const { error: executionError } = await supabase.from('test_executions').insert(caseIds.map((id) => ({ test_run_id: (run as any).id, test_case_id: id, organization_id: (project as any).organization_id, tester_id: assignedTo, status: 'not_run' as const })));
  if (executionError) {
    await supabase.from('test_runs').delete().eq('id', (run as any).id);
    return NextResponse.json({ error: 'Could not initialize test executions for this run.' }, { status: 500 });
  }
  return NextResponse.json({ testRun: run, testCaseCount: caseIds.length }, { status: 201 });
}

import { getPermissions } from './permissions';
import type { UserPermissions } from './permissions';
import type { UserRole } from '@/types/supabase';

// P0: enforce RBAC per-resource org, not "first org of user".
// Existing checkPermission() in rbac.ts uses currentOrg = memberships[0],
// which is wrong when a user belongs to multiple orgs.
export async function getRoleForOrg(supabase: any, userId: string, organizationId: string): Promise<UserRole | null> {
  const { data, error } = await supabase.from('organization_members').select('role').eq('organization_id', organizationId).eq('user_id', userId).maybeSingle();
  if (error || !data) return null;
  return data.role as UserRole;
}
export async function requireOrgPermission(supabase: any, userId: string, organizationId: string, permission: keyof Omit<UserPermissions, 'role'>): Promise<{ role: UserRole; permissions: UserPermissions }> {
  const role = await getRoleForOrg(supabase, userId, organizationId);
  if (!role) throw Object.assign(new Error('You do not have access to this organization.'), { status: 403 });
  const permissions = getPermissions(role);
  if (!permissions[permission]) throw Object.assign(new Error('Insufficient permissions.'), { status: 403 });
  return { role, permissions };
}
export function permissionError(e: unknown) {
  const status = (e as any)?.status ?? 500;
  const message = e instanceof Error ? e.message : 'Could not complete this action.';
  return { status, message };
}
